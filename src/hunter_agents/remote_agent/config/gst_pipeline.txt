# GStreamer AGX 硬件 H264 编码 → RTMP 推流流水线（V1.21.0 架构，文档 13.2.2 改版）
# D435 RGB 1280x720@30fps → nvvidconv(硬件转换) → nvv4l2h264enc(硬件编码) → flvmux → rtmpsink(SRS 1935)
# 编码参数：bitrate=3000000(3Mbps CBR), iframeinterval=30(1秒1关键帧), control-rate=1
# 会话制：收到 remote_control boot 帧才起流；流名 hunter_HUNTER-001_{session_id}（前缀=平台 RC_SRS_STREAM_PREFIX）
# 编码延迟：< 50ms（硬件编码）；浏览器侧经 WHEP(:9090) 拉流，车端不参与 WebRTC 信令
# 依赖：gstreamer1.0-plugins-good（flvmux/rtmpsink）；Jetson 默认已装，缺失时：
#   sudo apt-get install -y gstreamer1.0-plugins-good

v4l2src device=/dev/video0 ! \
  video/x-raw,width=1280,height=720,framerate=30/1 ! \
  videoconvert ! nvvidconv ! 'video/x-raw(memory:NVMM),format=NV12' ! \
  nvv4l2h264enc bitrate=3000000 iframeinterval=30 control-rate=1 profile=4 preset-level=1 ! \
  h264parse ! flvmux streamable=true ! \
  rtmpsink location="rtmp://192.168.31.35:1935/remote_control/hunter_HUNTER-001_{session_id}"

# 手工联调（不经 boot 帧，直接验 SRS 可达性与出画）：
#   { ping -c2 192.168.31.35 >/dev/null && ss -lut | grep -q ':1935' ; }   # 车→SRS 1935 可达
#   gst-launch-1.0 -e <上面的流水线（{session_id} 换成真实会话 ID 或自定测试流名）>
#   浏览器侧经验证 WHEP：POST http://192.168.31.35:9090/rtc/v1/whep/?app=remote_control&stream=hunter_HUNTER-001_{session_id}
